const STORE_KEY = "sdmz_shang";
const ALL = [];
DATA.forEach(s => s.questions.forEach(q => ALL.push(q)));

let mode = "level", levelIdx = 0, qIdx = 0, score = 0, lives = 3, timeLeft = 40, timerId = null, pool = [];

function $(id){ return document.getElementById(id); }
function show(id){ ["screen-home","screen-levels","screen-quiz","screen-result"].forEach(s => $(s).style.display = s===id?"block":"none"); window.scrollTo(0,0); }
function goHome(){ clearInterval(timerId); show("screen-home"); }
function loadStore(){ try{ return JSON.parse(localStorage.getItem(STORE_KEY))||{} }catch(e){ return {} } }
function saveStore(o){ try{ localStorage.setItem(STORE_KEY, JSON.stringify(o)) }catch(e){} }

/* ---------- 闯关 ---------- */
function showLevels(){
  const st = loadStore(); const g = $("levelGrid"); g.innerHTML = "";
  for(let i=0;i<10;i++){
    const stars = st["lv"+i]||0;
    const b = document.createElement("button");
    b.className="level-btn";
    b.innerHTML = "第"+(i+1)+"关<br><span class='stars'>"+"⭐".repeat(stars)+"☆".repeat(10-stars).slice(0,10-stars>0?10-stars:0)+"</span>";
    b.onclick = ()=>startLevel(i);
    g.appendChild(b);
  }
  show("screen-levels");
}
function startLevel(i){
  mode="level"; levelIdx=i; qIdx=0; score=0;
  pool = ALL.slice(i*10, i*10+10);
  $("timerTrack").style.display="none";
  renderQ(); show("screen-quiz");
}
/* ---------- 极限挑战 ---------- */
function startChallenge(){
  mode="challenge"; qIdx=0; score=0; lives=3; timeLeft=40;
  pool = ALL.slice().sort(()=>Math.random()-.5);
  $("timerTrack").style.display="block";
  renderQ(); show("screen-quiz");
  clearInterval(timerId);
  timerId = setInterval(()=>{
    timeLeft = Math.max(0, timeLeft - 0.2);
    $("timerFill").style.width = (timeLeft/40*100)+"%";
    $("statusLeft").textContent = "⏱ "+Math.ceil(timeLeft)+"秒";
    if(timeLeft<=0){ endChallenge(); }
  },200);
}
/* ---------- 出题 ---------- */
function renderQ(){
  const q = pool[qIdx % pool.length];
  if(mode==="level"){
    $("statusLeft").textContent = "第"+(levelIdx+1)+"关 · 第"+(qIdx+1)+"/10题";
    $("statusRight").textContent = "⭐ "+score;
  }else{
    $("statusLeft").textContent = "⏱ "+Math.ceil(timeLeft)+"秒";
    $("statusRight").innerHTML = "<span class='hearts'>"+"❤️".repeat(lives)+"</span> ⭐"+score;
  }
  $("qText").textContent = q.q;
  const box = $("opts"); box.innerHTML="";
  const letters=["A","B","C","D"];
  q.opts.forEach((o,i)=>{
    const b=document.createElement("button");
    b.className="opt";
    b.innerHTML="<span class='tag'>"+letters[i]+"</span>"+o;
    b.onclick=()=>answer(i,b,q);
    box.appendChild(b);
  });
  $("expPanel").className="panel exp"; $("factPanel").className="panel fact";
  $("nextBtn").style.display="none";
}
function answer(i, btn, q){
  const opts = document.querySelectorAll(".opt");
  opts.forEach(o=>o.style.pointerEvents="none");
  const ok = i===q.ans;
  opts[q.ans].classList.add("correct");
  if(ok){
    score++;
    const s=document.createElement("div"); s.className="fly-star"; s.textContent="⭐";
    s.style.left="50%"; s.style.top="40%"; document.body.appendChild(s);
    setTimeout(()=>s.remove(),1000);
    opts.forEach((o,idx)=>{ if(idx!==q.ans) o.classList.add("dim"); });
  }else{
    btn.classList.add("wrong");
    if(mode==="challenge"){ lives--; }
  }
  if(mode==="challenge"){ $("statusRight").innerHTML="<span class='hearts'>"+"❤️".repeat(lives)+"</span> ⭐"+score; }
  $("expPanel").innerHTML="<div class='verdict'>"+(ok?"🎉 答对啦！+1颗星":"💪 别灰心！正确答案是 "+"ABCD"[q.ans])+"</div><b class='tagline'>📖 为什么？</b>"+q.exp;
  $("factPanel").innerHTML="<b class='tagline'>💡 冷知识</b>"+q.fact;
  $("expPanel").classList.add("show"); $("factPanel").classList.add("show");
  if(mode==="level"){
    $("nextBtn").style.display="block";
    $("nextBtn").textContent = qIdx===9 ? "看看本关成绩 →" : "下一题 →";
  }else{
    if(lives<=0){ setTimeout(endChallenge, 1600); }
    else setTimeout(()=>{ qIdx++; renderQ(); }, 1600);
  }
}
function nextQuestion(){
  if(mode!=="level") return;
  qIdx++;
  if(qIdx>=10){ endLevel(); } else { renderQ(); }
}
function endLevel(){
  const st = loadStore();
  st["lv"+levelIdx] = Math.max(st["lv"+levelIdx]||0, score);
  saveStore(st);
  $("resultTitle").textContent = score>=8 ? "太棒了！" : score>=5 ? "不错哦！" : "继续加油！";
  $("resultStars").textContent = "⭐".repeat(score)+"☆".repeat(10-score);
  $("resultScore").textContent = score*10+" 分";
  $("resultNote").textContent = "第"+(levelIdx+1)+"关：答对 "+score+"/10 题";
  $("resultBtn1").textContent = levelIdx<9 ? "挑战下一关 →" : "再玩一次";
  show("screen-result");
}
function resultAction1(){
  if(mode==="level"){ startLevel(levelIdx<9?levelIdx+1:levelIdx); }
  else{ startChallenge(); }
}
function endChallenge(){
  clearInterval(timerId);
  const st = loadStore();
  const best = st["best"]||0;
  if(score>best){ st["best"]=score; saveStore(st); }
  $("resultTitle").textContent = score>best ? "新纪录！" : "挑战结束！";
  $("resultStars").textContent = "⭐".repeat(Math.min(score,20));
  $("resultScore").textContent = score*10+" 分";
  $("resultNote").textContent = "答对 "+score+" 题 · 历史最佳 "+Math.max(best,score)*10+" 分";
  $("resultBtn1").textContent = "再来一次 ⚡";
  show("screen-result");
}
